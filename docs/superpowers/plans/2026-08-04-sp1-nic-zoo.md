# SP1 — NIC Zoo + `monolith-net` Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make every 1996–2026 network card loadable on The Monolith as Tier-2 (`=m`) modules, and add a `monolith-net` helper that reaches the pre-Plug-and-Play ISA cards which emit no modalias.

**Architecture:** Kernel `=m` NIC drivers ride the existing `modules_install → squashfs → coldplug` path (PCI self-loads by modalias). A standalone, unit-tested `/usr/bin/monolith-net` script plus an `S35netprobe` init step handle ISA cards: safe self-identifying drivers auto-load at boot, and a bounded `io=` sweep is opt-in. A QEMU NIC-model boot-test matrix proves both paths.

**Tech Stack:** Linux 6.12.80 kernel config (savedconfig-style full `.config`), Gentoo crossdev ebuild, POSIX `/bin/sh` (rootfs uses GNU coreutils, not busybox), Python 3 + pexpect (boot-test), GitHub Actions.

## Global Constraints

- Rootfs scripts are **POSIX `/bin/sh`** and all-static; no bashisms required at runtime. `modprobe`/`ip`/`dhcpcd`/`lsmod` come from `sys-apps/kmod` + `iproute2` + `dhcpcd` already in `world`.
- **Never ship firmware blobs.** Do not set `CONFIG_EXTRA_FIRMWARE`.
- **Kernel `.config` changes REQUIRE a kernel ebuild revbump** — Portage keys binpkgs on version+USE+CHOST, not `.config`; without a revbump the S3 cache serves the old kernel and no new module is built.
- Kernel release string (`uname -r`) is `6.12.80-i486-monolith` (`CONFIG_LOCALVERSION="-i486-monolith"`); modules install under `/lib/modules/6.12.80-i486-monolith/`.
- Boot-test cannot run on the authoring machine (no `qemu-system-*`); its authority is the CI log. Local checks for kernel config are intent-only; the CI build + boot-test are the real gate.
- The rootfs `configs/` tree is mounted at `/configs` in the build container (`CONFIGS_DIR=/configs`); files under `configs/` are readable by `build-rootfs.sh`.

```sh
# Unit tests for monolith-net. Stubs modprobe/ip/dhcpcd on PATH and points the
# script at a fake sysfs so no real hardware or root is required.
set -u
HERE=$(cd "$(dirname "$0")" && pwd)
SCRIPT="$HERE/../../configs/rootfs-overlay/usr/bin/monolith-net"
fails=0
check() { # desc, expected_substr, actual
    if printf '%s' "$3" | grep -qF "$2"; then echo "  ok: $1"
    else echo "  FAIL: $1 (wanted '$2')"; echo "----"; printf '%s\n' "$3"; echo "----"; fails=$((fails+1)); fi
}

setup() {
    TMP=$(mktemp -d); export SYS="$TMP/sys"; mkdir -p "$SYS"
    BIN="$TMP/bin"; mkdir -p "$BIN"
    export MONOLITH_NET_SYSCLASS="$SYS"
    # stub modprobe: creates eth0 only when the (driver[:io]) spec equals $WIN.
    cat > "$BIN/modprobe" <<'STUB'
#!/bin/sh
[ "$1" = "-r" ] && exit 0
drv=$1; shift; addr=""
for a in "$@"; do case "$a" in io=*) addr=${a#io=};; esac; done
spec="$drv"; [ -n "$addr" ] && spec="$drv:$addr"
if [ "$spec" = "${WIN:-}" ]; then mkdir -p "$SYS/eth0"; echo up > "$SYS/eth0/operstate"; exit 0; fi
exit 1
STUB
    printf '#!/bin/sh\nexit 0\n' > "$BIN/ip"
    printf '#!/bin/sh\nexit 0\n' > "$BIN/dhcpcd"
    chmod +x "$BIN"/*
    export PATH="$BIN:$PATH"
}
teardown() { rm -rf "$TMP"; unset WIN; }

# 1. status with no NIC prints the incantation
setup; out=$(WIN= sh "$SCRIPT" status 2>&1); check "status/no-nic shows incantation" "monolith-net probe" "$out"; teardown
# 2. status with a NIC lists it
setup; mkdir -p "$SYS/eth0"; echo up > "$SYS/eth0/operstate"
out=$(sh "$SCRIPT" status 2>&1); check "status lists eth0" "eth0" "$out"; teardown
# 3. autoprobe: 3c509 self-identifies -> NIC appears, no banner
setup; out=$(WIN=3c509 sh "$SCRIPT" autoprobe 2>&1)
check "autoprobe finds 3c509" "" "$out"  # no crash
[ -d "$SYS/eth0" ] && echo "  ok: autoprobe created eth0" || { echo "  FAIL: autoprobe made no eth0"; fails=$((fails+1)); }
teardown
# 4. autoprobe with nothing -> prints incantation
setup; out=$(WIN= sh "$SCRIPT" autoprobe 2>&1); check "autoprobe/none shows incantation" "monolith-net probe" "$out"; teardown
# 5. probe finds ne at a non-default address
setup; out=$(WIN=ne:0x320 sh "$SCRIPT" probe 2>&1); check "probe finds ne io=0x320" "ne io=0x320" "$out"; teardown
# 6. probe finds nothing -> reports failure + incantation, exit 1
setup; out=$(WIN= sh "$SCRIPT" probe 2>&1); rc=$?
check "probe/none reports failure" "No ISA card answered" "$out"
[ "$rc" -eq 1 ] && echo "  ok: probe/none exits 1" || { echo "  FAIL: probe/none exit=$rc"; fails=$((fails+1)); }
teardown
# 7. incantation prints the liturgy
setup; out=$(sh "$SCRIPT" incantation 2>&1); check "incantation shows modprobe ne" "modprobe ne io=0x300" "$out"; teardown
# 8. unknown subcommand -> usage, exit 2
setup; out=$(sh "$SCRIPT" bogus 2>&1); rc=$?
[ "$rc" -eq 2 ] && echo "  ok: bad cmd exits 2" || { echo "  FAIL: bad cmd exit=$rc"; fails=$((fails+1)); }
teardown

echo; [ "$fails" -eq 0 ] && { echo "ALL PASS"; exit 0; } || { echo "$fails FAILED"; exit 1; }
```

- [ ] **Step 3: Write minimal implementation** — `configs/rootfs-overlay/usr/bin/monolith-net`

```sh
#!/bin/sh
# monolith-net — bring up network cards that can't announce themselves.
#
# PCI/virtual NICs coldplug automatically via their modalias. This handles the
# pre-Plug-and-Play ISA cards: 3c509 self-identifies (safe to load blindly),
# while ne/wd/lance need an explicit io= address. When auto-detection can't, it
# prints the incantation rather than hiding it.
set -u

SYSCLASS="${MONOLITH_NET_SYSCLASS:-/sys/class/net}"

list_ifaces() {
    for d in "$SYSCLASS"/*; do
        [ -e "$d" ] || continue
        n=${d##*/}
        [ "$n" = lo ] && continue
        echo "$n"
    done
}
have_nic() { [ -n "$(list_ifaces)" ]; }

bring_up() {
    ip link set "$1" up 2>/dev/null || true
    dhcpcd -b "$1" 2>/dev/null || true
}

print_incantation() {
    cat <<'BANNER'
No network card detected automatically.
ISA cards from before Plug-and-Play can't announce themselves — you tell the
kernel where the card lives:

    modprobe ne io=0x300        # NE2000 clone (try 0x280, 0x320, 0x340...)
    modprobe wd io=0x240        # SMC/WD 80x3
    modprobe lance io=0x300     # AMD LANCE

Or let the box try the common addresses for you:

    monolith-net probe
BANNER
}

cmd_status() {
    if have_nic; then
        echo "Network interfaces:"
        for i in $(list_ifaces); do
            s=$(cat "$SYSCLASS/$i/operstate" 2>/dev/null || echo unknown)
            echo "  $i ($s)"
        done
    else
        print_incantation
    fi
}

# Safe auto-probe: load only self-identifying ISA drivers. Called by S35netprobe.
cmd_autoprobe() {
    have_nic && return 0
    modprobe 3c509 2>/dev/null || true    # 3Com ID-port self-detect — safe
    have_nic && return 0
    print_incantation
    return 0
}

# Opt-in bounded blind sweep of the classic io= addresses.
cmd_probe() {
    if have_nic; then
        echo "A network interface already exists; nothing to probe."
        cmd_status
        return 0
    fi
    for spec in \
        ne:0x300 ne:0x280 ne:0x320 ne:0x340 ne:0x360 \
        wd:0x240 wd:0x300 wd:0x280 wd:0x2e0 wd:0x350 \
        lance:0x300 lance:0x320 lance:0x360 lance:0x3a0
    do
        drv=${spec%%:*}; addr=${spec##*:}
        echo "  trying $drv io=$addr ..."
        if modprobe "$drv" io="$addr" 2>/dev/null && have_nic; then
            iface=$(list_ifaces | head -n1)
            echo "  found $iface via $drv io=$addr"
            bring_up "$iface"
            return 0
        fi
        modprobe -r "$drv" 2>/dev/null || true   # bad guess leaves nothing resident
    done
    echo "No ISA card answered at the common addresses."
    print_incantation
    return 1
}

case "${1:-status}" in
    status|"")   cmd_status ;;
    autoprobe)   cmd_autoprobe ;;
    probe)       cmd_probe ;;
    incantation) print_incantation ;;
    *) echo "usage: monolith-net [status|probe|autoprobe|incantation]" >&2; exit 2 ;;
esac
```

```python
# NIC-model matrix: QEMU emulates much of the zoo. Coldplug models assert the
# driver loads by modalias; the ISA model (ne2k_isa, no modalias) asserts the
# monolith-net probe path instead.
NIC_MODEL_MODULE = {
    "pcnet": r"^pcnet32\b",
    "rtl8139": r"^8139(too|cp)\b",
    "tulip": r"^tulip\b",
    "ne2k_pci": r"^ne2k_pci\b",
}

def smoke_nic_model_is_module(child, model):
    pattern = NIC_MODEL_MODULE[model]
    return run_check(
        child, f"{model} NIC driver is loaded as a module (coldplug)", "lsmod",
        regex_matches(pattern, re.MULTILINE),
    )

def smoke_isa_probe(child, results):
    """
    ne2k_isa has no modalias, so coldplug must NOT have loaded `ne`. Then
    `monolith-net probe` sweeps the common io= addresses (QEMU's ne2k_isa sits
    at the default 0x300), after which `ne` is resident and an interface exists.
    """
    results.append(("ne NOT auto-loaded for ISA card (no modalias)",
                    *run_check(child, "ne not coldplugged", "lsmod",
                               regex_absent(r"^ne\b", re.MULTILINE))))
    results.append(("monolith-net probe finds the ISA NE2000",
                    *run_check(child, "monolith-net probe", "monolith-net probe",
                               contains("found"), timeout=60)))
    results.append(("ne driver resident after probe",
                    *run_check(child, "ne loaded post-probe", "lsmod",
                               regex_matches(r"^ne\b", re.MULTILINE))))
```

```python
def build_nic_cmd(args):
    # BIOS/ISOLINUX on the i440fx `pc` machine (has an ISA bus, needed for
    # ne2k_isa). Legacy -net form replaces QEMU's default NIC with exactly one
    # card of the requested model; works for pcnet/rtl8139/tulip/ne2k_pci/ne2k_isa.
    if not args.nic_model:
        raise BootTestError("--mode nic requires --nic-model")
    cmd = build_bios_cmd(args)
    cmd += ["-net", f"nic,model={args.nic_model}", "-net", "user"]
    return cmd

def run_nic(child, args):
    select_isolinux_label(child, "serial")
    for ms, desc in [
        (MILESTONE_INIT_START, "initramfs /init started"),
        (MILESTONE_OVERLAY_READY, "squashfs+overlay mounted"),
        (MILESTONE_EXEC_INIT, "pivot_root complete, executing /sbin/init"),
        (MILESTONE_RCS_START, "sysvinit rcS started"),
        (MILESTONE_RCS_COMPLETE, "sysvinit rcS completed"),
    ]:
        expect_milestone(child, ms, args.boot_timeout, desc)
    wait_for_shell(child)

    results = []
    if args.nic_model == "ne2k_isa":
        smoke_isa_probe(child, results)
    else:
        results.append((f"{args.nic_model} coldplug", *smoke_nic_model_is_module(child, args.nic_model)))
    results.append(("uname -r matches expected kernel", *smoke_kernel_version(child, args.kernel_version)))
    results.append(("overlay root is mounted", *smoke_overlay_mount(child)))

    ok = report_results(results)
    poweroff_and_wait(child)
    return ok
```
